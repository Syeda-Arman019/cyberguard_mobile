import 'package:flutter/material.dart';

class CyberColors {
  CyberColors._();

  // Backgrounds — deep navy / dark blue
  static const Color bgDark = Color(0xFF04060F); // blue-black
  static const Color bgSurface = Color(0xFF080F24); // dark navy
  static const Color cardBg = Color(0xFF0B1330); // glassy dark blue card
  static const Color cardBgElevated = Color(0xFF101A3E);

  // Borders & Dividers — thin light blue
  static const Color border = Color(0x336FB1FF); // #6FB1FF at 20%
  static const Color borderSubtle = Color(0x220E1F4D);

  // Accents — blue family
  static const Color cyan = Color(0xFF2F6BFF); // primary blue (accent token)
  static const Color purple = Color(0xFF4C8DFF); // bright blue (secondary accent)
  static const Color blue = Color(0xFF6FB1FF); // sky blue
  static const Color skyBlue = Color(0xFF6FB1FF);
  static const Color brightBlue = Color(0xFF4C8DFF);
  static const Color cyanHighlight = Color(0xFF22C7F0); // small use only

  // Button gradient (primary blue -> bright blue), white text
  static const Color buttonGradientStart = Color(0xFF2F6BFF);
  static const Color buttonGradientEnd = Color(0xFF4C8DFF);
  static const Color buttonTextColor = Colors.white;

  static const LinearGradient primaryButtonGradient = LinearGradient(
    colors: [Color(0xFF2F6BFF), Color(0xFF4C8DFF)],
    begin: Alignment.centerLeft,
    end: Alignment.centerRight,
  );

  // Hero card gradient (dashboard header)
  static const LinearGradient heroGradient = LinearGradient(
    colors: [Color(0xFF2F6BFF), Color(0xFF5FA8FF)],
    begin: Alignment.topLeft,
    end: Alignment.bottomRight,
  );

  // Risk Indicators — semantic colors kept, tuned for dark blue bg
  static const Color safe = Color(0xFF34D399);
  static const Color suspicious = Color(0xFFF5B942);
  static const Color malicious = Color(0xFFF0605D);

  // Typography
  static const Color textPrimary = Colors.white;
  static const Color textSecondary = Color(0xFF9FB2D9); // muted blue-grey
  static const Color textMuted = Color(0xFF6E82AC);

  // Gradients — vertical: dull navy blue at top -> blue-black at bottom
  // (muted/desaturated, not bright)
  static const LinearGradient backgroundGradient = LinearGradient(
    colors: [Color(0xFF0D1738), Color(0xFF080F24), Color(0xFF04060F)],
    begin: Alignment.topCenter,
    end: Alignment.bottomCenter,
  );

  static const LinearGradient cardGradient = LinearGradient(
    colors: [Color(0xFF101A3E), Color(0xFF080F24)],
    begin: Alignment.topLeft,
    end: Alignment.bottomRight,
  );
}

class CyberTheme {
  CyberTheme._();

  static ThemeData get darkTheme {
    return ThemeData.dark().copyWith(
      scaffoldBackgroundColor: CyberColors.bgDark,
      primaryColor: CyberColors.cyan,
      colorScheme: const ColorScheme.dark(
        primary: CyberColors.cyan,
        secondary: CyberColors.purple,
        surface: CyberColors.cardBg,
        error: CyberColors.malicious,
        onPrimary: Colors.white,
        onSurface: CyberColors.textPrimary,
      ),
      appBarTheme: const AppBarTheme(
        backgroundColor: Colors.transparent,
        elevation: 0,
        centerTitle: true,
        titleTextStyle: TextStyle(
          color: CyberColors.textPrimary,
          fontSize: 18,
          fontWeight: FontWeight.bold,
          letterSpacing: 0.5,
        ),
        iconTheme: IconThemeData(color: CyberColors.skyBlue),
      ),
      cardTheme: CardThemeData(
        color: CyberColors.cardBg,
        elevation: 2,
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(20),
          side: const BorderSide(color: CyberColors.border, width: 1),
        ),
      ),
      dialogTheme: DialogThemeData(
        backgroundColor: CyberColors.cardBg,
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(20),
          side: const BorderSide(color: CyberColors.border, width: 1.2),
        ),
        titleTextStyle: const TextStyle(
          color: CyberColors.textPrimary,
          fontSize: 18,
          fontWeight: FontWeight.bold,
        ),
        contentTextStyle: const TextStyle(
          color: CyberColors.textSecondary,
          fontSize: 14,
        ),
      ),
      elevatedButtonTheme: ElevatedButtonThemeData(
        style: ElevatedButton.styleFrom(
          backgroundColor: CyberColors.cyan,
          foregroundColor: CyberColors.buttonTextColor,
          shape: const StadiumBorder(),
          textStyle: const TextStyle(
            fontWeight: FontWeight.w600,
            fontSize: 15,
            letterSpacing: 0.5,
          ),
        ),
      ),
      outlinedButtonTheme: OutlinedButtonThemeData(
        style: OutlinedButton.styleFrom(
          foregroundColor: CyberColors.skyBlue,
          side: const BorderSide(color: CyberColors.border, width: 1.2),
          shape: const StadiumBorder(),
          textStyle: const TextStyle(
            fontWeight: FontWeight.w600,
            fontSize: 14,
          ),
        ),
      ),
      navigationBarTheme: NavigationBarThemeData(
        backgroundColor: CyberColors.bgSurface,
        indicatorColor: CyberColors.brightBlue,
        iconTheme: WidgetStateProperty.resolveWith((states) {
          return IconThemeData(
            color: states.contains(WidgetState.selected)
                ? Colors.white
                : CyberColors.textSecondary,
          );
        }),
        labelTextStyle: WidgetStateProperty.resolveWith((states) {
          return TextStyle(
            fontSize: 11.5,
            fontWeight: states.contains(WidgetState.selected)
                ? FontWeight.bold
                : FontWeight.w500,
            color: states.contains(WidgetState.selected)
                ? Colors.white
                : CyberColors.textSecondary,
          );
        }),
      ),
      switchTheme: SwitchThemeData(
        thumbColor: WidgetStateProperty.resolveWith((states) {
          return states.contains(WidgetState.selected)
              ? Colors.white
              : CyberColors.textMuted;
        }),
        trackColor: WidgetStateProperty.resolveWith((states) {
          return states.contains(WidgetState.selected)
              ? CyberColors.brightBlue
              : CyberColors.bgSurface;
        }),
      ),
      dividerTheme: const DividerThemeData(color: CyberColors.border),
      inputDecorationTheme: const InputDecorationTheme(
        border: OutlineInputBorder(
          borderRadius: BorderRadius.all(Radius.circular(14)),
          borderSide: BorderSide(color: CyberColors.border),
        ),
        enabledBorder: OutlineInputBorder(
          borderRadius: BorderRadius.all(Radius.circular(14)),
          borderSide: BorderSide(color: CyberColors.border),
        ),
        focusedBorder: OutlineInputBorder(
          borderRadius: BorderRadius.all(Radius.circular(14)),
          borderSide: BorderSide(color: CyberColors.brightBlue, width: 1.5),
        ),
      ),
    );
  }
}
